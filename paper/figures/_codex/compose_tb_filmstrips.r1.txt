    49	    inconsistent, so rendering BOTH localises noise: ric reads per-frame positions (ch0:3),
    50	    fk rebuilds them from rotations (ch3:9) over the bone chain.

    52	    if norm_txjc.shape[-1] != 13:

    54	        # (17/18 channels) -- that path would silently misread the channel layout.
    55	        raise ValueError(f"legacy world_of expects 13 channels, got {norm_txjc.shape[-1]} -- "
    56	                         f"KTJD tensors must go through world_of_ktjd")

    58	    if recover == "fk":
    59	        return recover_from_bvh_rot_np(raw, parents, offsets)        # [T,J,3]
    60	    return _recover_world_positions(raw)                             # [T,J,3]

    63	def world_of_ktjd(norm_seg18, base, rig, strict_gt):
    64	    """KTJD-17 world positions via the OFFICIAL codec, both paths at once.
    65	
    66	    norm_seg18 [T,J,18] normalized (plane 17 = heading flag, sliced off); de-normalized with the
    67	    adapter's exact std trick, then decode_ktjd17 (direct + FK, NO temporal integration --
    68	    velocity channels are never integrated, per the KTJD contract). strict_gt=False for model
    69	    output (degenerate predicted 6D tolerated and reported by the codec).
    70	    Returns (direct [T,J,3], fk [T,J,3]).

    72	    from src.data.ktjd17.decoder import decode_ktjd17
    73	    sk = base._skeleton(rig)
    74	    J = norm_seg18.shape[1]
    75	    # per-cell mean/std (2026-08-21): the served/predicted tensor is standardized per
    76	    # (rig, joint, channel), so de-normalization is the repo convention x*(std+floor)+mean with
    77	    # BOTH taken from the stats artifact (codex round-S7 blocker 2: this used the superseded
    78	    # scale-only path and called a function that no longer exists).

    89	def one_euro(x, fps=20.0, min_cutoff=1.5, beta=0.3, d_cutoff=1.0):
    90	    """One-euro filter along axis 0 of x [T, ...] (Casiez et al. 2012), the game-industry
    91	    standard adaptive low-pass: strong smoothing at low speeds (kills uniform micro-jitter on
    92	    near-static joints -- the measured failure mode, see jitter_analysis_s32.txt), weak at high
    93	    speeds (bursts stay sharp). Pure post-process on the MODEL OUTPUT channels; deployment
    94	    candidate (user 2026-08-28 plan b)."""
    95	    import numpy as _np
    96	    def alpha(cutoff):
    97	        tau = 1.0 / (2 * _np.pi * cutoff)
    98	        te = 1.0 / fps
    99	        return 1.0 / (1.0 + tau / te)
   100	    y = x.copy()
   101	    dx_prev = _np.zeros_like(x[0])
   102	    for t in range(1, x.shape[0]):
   103	        dx = (x[t] - y[t - 1]) * fps
   104	        a_d = alpha(d_cutoff)
   105	        dx_hat = a_d * dx + (1 - a_d) * dx_prev
   106	        cutoff = min_cutoff + beta * _np.abs(dx_hat)
   107	        a = alpha(cutoff)
   108	        y[t] = a * x[t] + (1 - a) * y[t - 1]
   109	        dx_prev = dx_hat
   110	    return y

   117	\section{Adapter settings}\label{app:lora}
   118	
   119	\paragraph{Data view of a new rig.} The
   120	data view of a new rig is built by the corpus script with three additions: joints are pruned to the
   121	deformation-bone convention of the training library (end markers, cosmetic and helper joints
   122	dropped by name or description, never the root, heading carrier, trunk or wing joints, and only
   123	leaf-closed subtrees so no forward kinematics is re-run); captions are rewritten to the corpus style
   124	``The $\langle$species$\rangle$ $\langle$action$\rangle$''; and the train/validation split is made
   125	by source animation so that no clip straddles it. Training follows the backbone recipe with the
   126	learning rate scaled linearly to the batch, one pass over every clip per epoch, 300--900 epochs and
   127	a shortened forward-kinematics warm-up. A single adapter can also be shared by a group of rigs; we
   128	report this as a negative result.
   129	
   130	\paragraph{Probing text generalisation.} To test whether an adapted rig still follows language it
   131	was not adapted on, we render captions of related species that no checkpoint in the adapter's
   132	initialisation lineage trained on, and authored out-of-distribution captions, on the adapted rig;
   133	the probe walks the lineage by content hash and refuses any caption seen upstream.
   134	
   135	
   136	
   137	\paragraph{Hyper-parameters.} Table~\ref{tab:lora_hp} lists the per-rig settings. The learning rate
   138	follows the linear-scaling rule from $10^{-4}$ at batch 8; the batch is the largest divisor of the
